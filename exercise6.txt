Declare Real prices[10]
For 0 to 9
    Display "Enter the 10 prices."
    Input prices[i]
End For

Set Highest = prices[0]
Set Lowest = prices[0]
Set Total = 0

For 0 to 9
    Set Total = Total + prices[i]
    If prices[i] > Highest Then
    Set Highest = prices[i]
End If

If prices[i] < Lowest Then
    Set Lowest = prices[i]
End If

End For

Set Average = Total / 10
Display Highest
Display Lowest
Display Average

For 0 to 9
    If prices[i] > Average
        Display prices[i]
